export function telInter(t: string): string {
  let d = String(t || "").replace(/\D/g, "");
  if (d.startsWith("00")) d = d.slice(2);
  else if (d.startsWith("0")) d = "32" + d.slice(1);
  return d.length >= 10 && d.length <= 15 ? d : "";
}
function bruxelles(d = new Date()) {
  const p = Object.fromEntries(
    new Intl.DateTimeFormat("en-GB", {
      timeZone: "Europe/Brussels", year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", hour12: false,
    }).formatToParts(d).map((x) => [x.type, x.value]),
  );
  return { iso: `${p.year}-${p.month}-${p.day}`, heure: Number(p.hour) % 24, an: Number(p.year) };
}

async function secret(nom: string) {

async function hmac(msg: string) {
  const k = await crypto.subtle.importKey("raw", new TextEncoder().encode(await secret("signature")), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  const s = await crypto.subtle.sign("HMAC", k, new TextEncoder().encode(msg));
  return btoa(String.fromCharCode(...new Uint8Array(s))).replace(/[+/=]/g, "");
}

// ---------- SMS (Brevo) ----------
export async function envoyerSMS(tel: string, texte: string, cle?: string) {
  const dest = telInter(tel);
  if (!dest || !texte.trim()) return { ok: false, raison: "numéro ou texte vide" };
  if (cle) {
    const { data } = await db.from("sms_log").select("id").eq("cle", cle).maybeSingle();
    if (data) return { ok: true, deja: true };
  }
  const key = Deno.env.get("BREVO_API_KEY");
  let ok = false, reponse = "pas de clé Brevo";
  if (key) {
    const corps = JSON.stringify({
      sender: (Deno.env.get("SMS_EXPEDITEUR") || "RefletAmbre").slice(0, 11),
      recipient: dest, content: texte.slice(0, 600), type: "transactional", unicodeEnabled: /[^\x00-\x7F]/.test(texte),
    });
    const h = { "api-key": key, "content-type": "application/json", accept: "application/json" };
    let r = await fetch("https://api.brevo.com/v3/transactionalSMS/send", { method: "POST", headers: h, body: corps });
    if (r.status === 404) r = await fetch("https://api.brevo.com/v3/transactionalSMS/sms", { method: "POST", headers: h, body: corps });
    ok = r.ok; reponse = (await r.text()).slice(0, 400);
  }
  await db.from("sms_log").insert({ cle: cle || null, tel: dest, texte, ok, reponse });
  return { ok, raison: ok ? "" : reponse };
}

async function vapid() {
  let v: any = await reglage("vapid");
  if (!v?.publicKey || !v?.privateKey) {
    v = webpush.generateVAPIDKeys();
    await db.from("reglages").upsert({ cle: "vapid", valeur: v, maj: new Date().toISOString() });
  }
  return v;
}
export async function pushDori(titre: string, texte: string) {
  const abos: any[] = ((await reglage("push_abos")) as any[]) || [];
  if (!abos.length) return { ok: false, raison: "aucun téléphone activé" };
  const v = await vapid();
  const garde: any[] = [], statuts: any[] = [];
  let envoyes = 0;
  for (const a of abos) {
    try {
      const d: any = webpush.generateRequestDetails(a, JSON.stringify({ titre, texte, url: "https://refletambre.be/#panel" }), {
        vapidDetails: { subject: "mailto:refletambre@outlook.be", publicKey: v.publicKey, privateKey: v.privateKey },
        TTL: 86400, urgency: "high",
      });
      const r = await fetch(d.endpoint, { method: "POST", headers: d.headers, body: d.body });
      statuts.push(r.status);
      if (r.status === 404 || r.status === 410) continue; // téléphone désabonné : on l'oublie
      if (r.ok) envoyes++;
    } catch (e) { statuts.push(String(e)); }
    garde.push(a);
  }
  if (garde.length !== abos.length) await db.from("reglages").upsert({ cle: "push_abos", valeur: garde, maj: new Date().toISOString() });
  console.log("push", JSON.stringify(statuts));
  return { ok: envoyes > 0, envoyes, telephones: abos.length, statuts };
}

async function verifier(req: Request, body: any) {
  const t = String(body.jeton || "");
  const [p, sig] = t.split(".");
  if (!p || !sig || (await hmac(p)) !== sig) return null;
  const o = JSON.parse(atob(p));
  if (o.exp < Date.now()) return null;
  return o as { a: number; exp: number };
}

async function charger() {

    db.from("clientes").select("data"),
    db.from("rdv").select("data,recu").gte("date", addJ(bruxelles().iso, -366)),

    const { data: ancien } = await db.from("rdv").select("date,heure").eq("id", id).maybeSingle();
    const row: any = { id, cid: Number(r.cid), date: r.date, heure: r.heure, statut: r.statut, data: r, maj: new Date().toISOString() };
    if (!ancien && r.recu) row.recu = new Date(r.recu).toISOString();
    if (ancien && (ancien.date !== r.date || ancien.heure !== r.heure)) row.rappel_envoye = false;
    const { error } = await db.from("rdv").upsert(row);
    if (error) return erreur("rdv: " + error.message, 500);
  }
  if (Array.isArray(b.supprRdv) && b.supprRdv.length) await db.from("rdv").delete().in("id", b.supprRdv.map(Number));
  if (Array.isArray(b.supprC) && b.supprC.length) await db.from("clientes").delete().in("id", b.supprC.map(Number));
  if (b.S && typeof b.S === "object") await db.from("reglages").upsert({ cle: "panel", valeur: b.S, maj: new Date().toISOString() });
  return rep({ ok: true });
}

async function emails(b: any) {
  const l = Array.isArray(b.envois) ? b.envois.slice(0, 200) : [];
  let ok = 0, ko = 0, raison = "";
  for (const e of l) {
    const r = await envoyerEmail(txt(e.to, 120), txt(e.sujet, 150), txt(e.texte, 4000), e.cle ? txt(e.cle, 120) : undefined);
    if (r.ok) ok++; else { ko++; raison = r.raison || raison; }
  }
  return rep({ ok: true, envoyes: ok, rates: ko, raison });
}
async function sms(b: any) {
  const l = Array.isArray(b.envois) ? b.envois.slice(0, 400) : [];
  let ok = 0, ko = 0, raison = "";
  for (const e of l) {
    const r = await envoyerSMS(txt(e.tel, 30), txt(e.texte, 600), e.cle ? txt(e.cle, 120) : undefined);
    if (r.ok) ok++; else { ko++; raison = r.raison || raison; }
  }
  return rep({ ok: true, envoyes: ok, rates: ko, raison });
}

function paques(y: number) {
  const a = y % 19, b = Math.floor(y / 100), c = y % 100, d = Math.floor(b / 4), e = b % 4, f = Math.floor((b + 8) / 25), g = Math.floor((b - f + 1) / 3),
    h = (19 * a + b - d - g + 15) % 30, i = Math.floor(c / 4), k = c % 4, l = (32 + 2 * e + 2 * i - h - k) % 7, m = Math.floor((a + 11 * h + 22 * l) / 114);
  const mo = Math.floor((h + l - 7 * m + 114) / 31), j = ((h + l - 7 * m + 114) % 31) + 1;
  return `${y}-${String(mo).padStart(2, "0")}-${String(j).padStart(2, "0")}`;
}
const nDim = (y: number, mo: number, n: number) => {
  const d = new Date(Date.UTC(y, mo, 1)); const off = (7 - d.getUTCDay()) % 7; d.setUTCDate(1 + off + 7 * (n - 1)); return d.toISOString().slice(0, 10);
};
const p2 = (n: number) => String(n).padStart(2, "0");
const FETES: Record<string, (y: number) => string> = {
  nouvelan: (y) => `${y}-01-01`, valentin: (y) => `${y}-02-14`, femme: (y) => `${y}-03-08`, paques,
  meres: (y) => nDim(y, 4, 2), peres: (y) => nDim(y, 5, 2), nationale: (y) => `${y}-07-21`, nicolas: (y) => `${y}-12-06`, noel: (y) => `${y}-12-25`,
};

async function taches() {
  const S = (await panelS()) || {};
  const { iso: auj, heure, an } = bruxelles();
  const fait: string[] = [];
  const auto = S.auto || {};
  const { data: CL } = await db.from("clientes").select("data");
  const clientes = (CL || []).map((x: any) => x.data).filter((c: any) => telInter(c.tel));
  const perso = (t: string, c: any, extra: Record<string, string> = {}) =>
    Object.entries({ prenom: c.prenom || "", annee: String(an), code: c.tel || "", ...extra }).reduce((s, [k, v]) => s.replaceAll(`{${k}}`, v), t);

  // 1. Relance de Dori : demandes sans réponse depuis X heures
  if (S.relance?.on !== false) {
    const h = Number(S.relance?.h || 24);
    const { data: att } = await db.from("rdv").select("id,data").eq("statut", "attente").eq("relance_envoyee", false).lt("recu", new Date(Date.now() - h * 3600e3).toISOString());
    if (att?.length) {
      const n = att.length;
      const r = await pushDori(`⏰ ${n} demande${n > 1 ? "s attendent" : " attend"} votre réponse`, `Depuis plus de ${h} h. Touchez pour ouvrir votre panel et valider ♡`);
      if (r.ok) await db.from("rdv").update({ relance_envoyee: true }).in("id", att.map((x: any) => x.id));
      fait.push(`relance:${n}`);
    }
  }
  // 2. Rappel la veille à 18:00
  if (SMS_AUTO && heure >= 18 && auto.rappel?.on !== false) {
    const demain = addJ(auj, 1);
    const { data: l } = await db.from("rdv").select("id,cid,heure,data").eq("date", demain).eq("statut", "confirme").eq("rappel_envoye", false);
    for (const r of l || []) {
      const c = clientes.find((x: any) => Number(x.id) === Number(r.cid));
      if (!c) continue;
      const t = perso(auto.rappel?.txt || "Bonjour {prenom}, petit rappel : votre RDV chez Reflet Ambré est demain à {heure}. Un empêchement ? 0475 72 18 52. À demain ♡ Dori", c, { heure: r.heure });
      const s = await envoyerSMS(c.tel, t, `rappel-${r.id}-${demain}`);
      if (s.ok) await db.from("rdv").update({ rappel_envoye: true }).eq("id", r.id);
    }
    fait.push("rappels");
  }
  // 3. Fêtes de Belgique à 10:00
  const dest = clientes.filter((c: any) => !(c.statut === "mauvais" && !S.autoBad));
  if (SMS_AUTO && heure >= 10) {
    for (const [k, q] of Object.entries(FETES)) {
      if (q(an) !== auj || !auto[k]?.on || !auto[k]?.txt) continue;
      for (const c of dest) await envoyerSMS(c.tel, perso(auto[k].txt, c), `${k}-${an}-${c.id}`);
      fait.push(k);
    }
  }
  // 4. Pour chaque cliente : J-30, jour J (8:00), anniversaire de mariage
  for (const c of (SMS_AUTO ? dest : [])) {
    const m = c.mariage;
    if (!m || !/^\d{4}-\d{2}-\d{2}$/.test(m)) continue;
    if (heure >= 10 && auto.j30?.on && addJ(m, -30) === auj) await envoyerSMS(c.tel, perso(auto.j30.txt, c), `j30-${c.id}-${m}`);
    if (heure >= 8 && auto.jourj?.on && m === auj) await envoyerSMS(c.tel, perso(auto.jourj.txt, c), `jourj-${c.id}-${m}`);
    if (heure >= 10 && auto.anniv?.on && m < auj && m.slice(5) === auj.slice(5)) await envoyerSMS(c.tel, perso(auto.anniv.txt, c), `anniv-${c.id}-${an}`);
  }
  // 5. Ménage : RDV de plus d'un an, vieux essais
  await db.from("rdv").delete().lt("date", addJ(auj, -366));
  await db.from("tentatives").delete().lt("quand", new Date(Date.now() - 3 * 864e5).toISOString());
  return rep({ ok: true, fait });
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  try {
    const u = new URL(req.url);
    const b: any = req.method === "POST" ? await req.json().catch(() => ({})) : {};
    const f = b.f || u.searchParams.get("f") || "site";

    if (f === "site") return rep(await site());
    if (f === "reserver") return await reserver(req, b);
    if (f === "test_notif") {
      const qui = ip(req);
      if (await trop("test_notif", qui, 5, 60)) return erreur("Trop d'essais", 429);
      await noter("test_notif", qui);
      return rep(await pushDori("Test du serveur Reflet Ambré", "Si Dori voit ce message, les notifications marchent ♡"));
    }
    if (f === "taches") {
      if ((b.secret || u.searchParams.get("secret")) !== (await secret("taches"))) return erreur("non", 403);
      return await taches();
    }
    if (f === "login") {
      const qui = ip(req);
      if (await trop("login", qui, 8, 15)) return erreur("Trop d'essais. Réessayez dans 15 minutes.", 429);
      const S = await panelS();
      const code = txt(b.code, 12);
      const acces = S?.acces?.length ? S.acces : [{ id: 1, nom: "Dori", code: CODE_DEPART, role: "tout" }];
      const ac = acces.find((a: any) => String(a.code) === code);
      if (!ac) { await noter("login", qui); return erreur("Code incorrect", 401); }
      const p = btoa(JSON.stringify({ a: ac.id, exp: Date.now() + 30 * 864e5 }));
      return rep({ ok: true, jeton: `${p}.${await hmac(p)}`, acces: { id: ac.id, nom: ac.nom, role: ac.role }, codeDepart: code === CODE_DEPART });
    }

    // tout le reste : il faut être connecté
    const qui = await verifier(req, b);
    if (!qui) return erreur("Session expirée, retapez votre code", 401);
    if (f === "charger") return await charger();
    if (f === "push_cle") return rep({ ok: true, cle: (await vapid()).publicKey });
    if (f === "push_abo") {
      const a = b.abo || {};
      if (typeof a.endpoint !== "string" || !a.endpoint.startsWith("https://") || !a.keys?.p256dh || !a.keys?.auth) return erreur("abonnement incorrect");
      const abo = { endpoint: txt(a.endpoint, 600), keys: { p256dh: txt(a.keys.p256dh, 200), auth: txt(a.keys.auth, 100) } };
      const l: any[] = (((await reglage("push_abos")) as any[]) || []).filter((x: any) => x.endpoint !== abo.endpoint);
      l.push(abo);
      await db.from("reglages").upsert({ cle: "push_abos", valeur: l.slice(-10), maj: new Date().toISOString() });
      return rep({ ok: true, telephones: Math.min(l.length, 10) });
    }
    if (f === "push_test") return rep(await pushDori("🔔 Test Reflet Ambré", "Les notifications marchent sur ce téléphone ♡"));
    if (f === "sync") return await sync(b);
    if (f === "sms") return await sms(b);
    if (f === "email") return await emails(b);
    if (f === "reglage") {
      const cle = txt(b.cle, 30);
      if (!["volume", "musiqueOn", "musique", "apropos"].includes(cle)) return erreur("réglage inconnu");
      await db.from("reglages").upsert({ cle, valeur: b.valeur ?? null, maj: new Date().toISOString() });
      return rep({ ok: true });
    }
    if (f === "envoi") { // adresse pour déposer une photo / vidéo / musique directement dans le stockage
      const ext = txt(b.ext, 5).replace(/[^a-z0-9]/gi, "").toLowerCase() || "bin";
      const chemin = `${txt(b.dossier, 20).replace(/[^a-z]/g, "") || "medias"}/${newId()}.${ext}`;
      const { data, error } = await db.storage.from("site").createSignedUploadUrl(chemin);
      if (error) return erreur(error.message, 500);
      return rep({ ok: true, chemin, envoi: data.signedUrl, url: urlPublic(chemin) });
    }
    if (f === "media_ajout") {
      const m = b.media || {};
      const { error } = await db.from("medias").upsert({ id: txt(m.id, 40), type: m.type === "video" ? "video" : "image", cat: txt(m.cat, 30), haut: !!m.haut, legende: txt(m.legende, 120), chemin: txt(m.chemin, 120), date: Number(m.date) || Date.now() });
      return error ? erreur(error.message, 500) : rep({ ok: true });
    }
    if (f === "media_suppr") {
      const { data } = await db.from("medias").select("chemin").eq("id", txt(b.id, 40)).maybeSingle();
      if (data) { await db.storage.from("site").remove([data.chemin]); await db.from("medias").delete().eq("id", txt(b.id, 40)); }
      return rep({ ok: true });
    }
    return erreur("action inconnue");
  } catch (e) {
    console.error(e);
    return erreur("Erreur serveur", 500);
  }
});
